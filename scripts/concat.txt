    $text = adb mdns services 2>&1 | Out-String

function Ensure-TorChatEnvironment {

            $bytes = New-Object byte[] 24
            $rng = [System.Security.Cryptography.RandomNumberGenerator]::Create()
            try { $rng.GetBytes($bytes) } finally { $rng.Dispose() }
            $values['TORCHAT_DATABASE_PASSWORD'] = [Convert]::ToBase64String($bytes).Replace('+','A').Replace('/','B').Replace('=','')
        }

        $rng = [System.Security.Cryptography.RandomNumberGenerator]::Create()

        $values['TORCHAT_PAIRING_SECRET'] = [Convert]::ToBase64String($bytes).Replace('+','A').Replace('/','B').Replace('=','')

    $utf8NoBom = New-Object System.Text.UTF8Encoding($false)

function Invoke-DockerComposeWithRetry {
    param([Parameter(Mandatory = $true)][string[]]$Arguments, [Parameter(Mandatory = $true)][string]$FailureMessage)
    for ($attempt = 1; $attempt -le 3; $attempt++) {
        & docker @Arguments
        if ($LASTEXITCODE -eq 0) { return }
        if ($attempt -lt 3) {
            $delay = 5 * $attempt
            Write-Warning "Docker command failed (attempt $attempt/3). Retrying in $delay seconds. Inspect the Docker output above: this may be a registry timeout or an image build error."
            Start-Sleep -Seconds $delay
        }
    }
    throw $FailureMessage
}

function Stop-ConflictingTorChatStacks {
    if ($KeepOtherStacks) { return }
    $currentProject = $environmentState.Values['TORCHAT_COMPOSE_PROJECT']

        Where-Object { $_ -ne $currentProject }

        if ($containers.Count -eq 0) { continue }

        & docker rm --force @containers | Out-Null

        Write-Progress -Id 1 -Activity 'TorChat: przygotowanie połączenia onion' -Status $progressStatus -PercentComplete $percent

        try {
            $probe = Invoke-OnionHealthProbe -Url $OnionUrl -SocksPort $SocksPort -MaxTimeSeconds 75
            if ($probe.ExitCode -ne 0) {
                $lastOnionProbeError = $probe.Output
            } else {
                $payloadText = $probe.Output
                $onionHealth = $payloadText | ConvertFrom-Json
                if (($null -ne $onionHealth) -and ($null -ne $onionHealth.PSObject.Properties['status'])) {
                    if ($onionHealth.status -eq 'ok') {
                        Write-Progress -Id 1 -Activity 'TorChat: przygotowanie połączenia onion' -Completed
                        return
                    }
                }
                $lastOnionProbeError = "Unexpected onion health payload: $payloadText"
            }
        } catch {
            $lastOnionProbeError = $_.Exception.Message
        }

        Write-Host "[torchat] Onion circuit is still warming up (attempt $attempt/$onionAttempts; each probe allows 75 s)."
        Start-Sleep -Seconds 2
    }

    Write-Progress -Id 1 -Activity 'TorChat: przygotowanie połączenia onion' -Completed
    $torLogs = Read-DockerServiceLogs -ComposeArgs $ComposeArgs -Service 'tor'
    $serverLogs = Read-DockerServiceLogs -ComposeArgs $ComposeArgs -Service 'server' -Tail 40
    throw "Tor onion healthcheck failed through SOCKS at $OnionUrl. Last probe: $lastOnionProbeError`n--- tor logs ---`n$torLogs`n--- server logs ---`n$serverLogs"
}

function Invoke-TorChat {
    param([string]$Script, [hashtable]$Parameters = @{})
    $scriptPath = Join-Path $PSScriptRoot $Script
    $pwsh = Get-Command pwsh -ErrorAction SilentlyContinue
    if ($pwsh) {
        $args = @('-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', $scriptPath)
        foreach ($entry in $Parameters.GetEnumerator()) {
            $args += "-$($entry.Key)"
            if ($entry.Value -isnot [switch] -and $entry.Value -isnot [bool]) {
                $args += [string]$entry.Value
            } elseif ([bool]$entry.Value) {
                # Switches are represented by presence only.
            } else {
                $args = $args[0..($args.Count - 2)]
            }
        }
        & $pwsh.Source @args
    } else {
        & $scriptPath @Parameters
    }
    if (-not $?) { throw "$Script failed." }
}

function Clear-TorChatDesktopState {
    $runtimePath = [IO.Path]::GetFullPath((Join-Path $repoRoot 'target\debug\torchat-desktop.exe'))
    $releasePath = [IO.Path]::GetFullPath((Join-Path $repoRoot 'target\release\torchat-desktop.exe'))
    $running = @(Get-CimInstance Win32_Process -Filter "Name='torchat-desktop.exe'" |
        Where-Object { $_.ExecutablePath -eq $runtimePath -or $_.ExecutablePath -eq $releasePath })
    foreach ($process in $running) {
        Write-Host "[torchat] Stopping desktop runtime PID $($process.ProcessId) before clearing state."
        Stop-Process -Id ([int]$process.ProcessId) -Force -ErrorAction SilentlyContinue
    }
    if ($running.Count -gt 0) { Start-Sleep -Milliseconds 300 }
    $clientRoot = Join-Path $repoRoot '.torchat\clients\desktop'
    $files = @(
        (Join-Path $clientRoot 'identity.key'),
        (Join-Path $clientRoot 'identity.state.db'),
        (Join-Path $clientRoot 'identity.state.db-wal'),
        (Join-Path $clientRoot 'identity.state.db-shm')
    )
    foreach ($file in $files) {
        if (Test-Path -LiteralPath $file) {
            for ($attempt = 1; $attempt -le 5; $attempt++) {
                try {
                    Remove-Item -LiteralPath $file -Force -ErrorAction Stop
                    break
                } catch {
                    if ($attempt -eq 5) { throw }
                    Start-Sleep -Milliseconds 250
                }
            }
        }
    }
    Write-Host '[torchat] Cleared desktop client identity and encrypted local state.'
}

        Stop-Process -Id ([int]$process.ProcessId) -Force -ErrorAction SilentlyContinue
    }
    if ($running.Count -gt 0) { Start-Sleep -Milliseconds 300 }
}
